# ---------------------------------------------------------------------------
# Vape.dll - self-contained injector.
# Embeds the built SSJJNative.dll as RCDATA 422 and injects it via
# LoadLibraryW (extracting to %TEMP% at runtime). Console subsystem
# (no WIN32 flag) so that wmain + console selector work.
# ---------------------------------------------------------------------------
# ---------------------------------------------------------------------------
# L3 resource encryption
# ---------------------------------------------------------------------------
# 1) Generate a random 32-byte key once (persisted so reconfigure does not
#    churn the build). Lives in the injector binary; hardened by L4 (VMP).
set(SSJJ_KEY_HEADER "${CMAKE_CURRENT_BINARY_DIR}/ssjj_key.h")
if(NOT EXISTS "${SSJJ_KEY_HEADER}")
    set(SSJJ_KEY_BYTES "")
    foreach(_i RANGE 0 31)
        string(RANDOM LENGTH 2 ALPHABET 0123456789abcdef SSJJ_KEY_BYTE)
        string(APPEND SSJJ_KEY_BYTES "0x${SSJJ_KEY_BYTE},")
    endforeach()
    file(WRITE "${SSJJ_KEY_HEADER}"
        "/* generated by CMake - do not edit */\n"
        "static const unsigned char SSJJ_KEY[32] = { ${SSJJ_KEY_BYTES} };\n")
    message(STATUS "Generated L3 encryption key: ${SSJJ_KEY_HEADER}")
endif()

# 2) Build-time encryptor tool (shared by nothing at runtime; kept in build)
add_executable(ssjj_encrypt encrypt_tool.c)
target_include_directories(ssjj_encrypt PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_compile_definitions(ssjj_encrypt PRIVATE _CRT_SECURE_NO_WARNINGS)
target_compile_options(ssjj_encrypt PRIVATE /O2 /MT /W3)
set_target_properties(ssjj_encrypt PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")

# 3) Encrypt SSJJNative.dll after it is built -> ssjjnative.enc (ciphertext)
set(SSJJ_NATIVE_DLL "${CMAKE_BINARY_DIR}/dist/SSJJNative.dll")
set(SSJJ_NATIVE_ENC "${CMAKE_CURRENT_BINARY_DIR}/ssjjnative.enc")
file(TO_CMAKE_PATH "${SSJJ_NATIVE_ENC}" SSJJ_NATIVE_DLL_RC)
add_custom_command(TARGET SSJJNative POST_BUILD
    COMMAND $<TARGET_FILE:ssjj_encrypt>
            $<TARGET_FILE:SSJJNative> "${SSJJ_NATIVE_ENC}"
    COMMENT "[L3] Encrypting SSJJNative.dll -> ssjjnative.enc"
    VERBATIM)
add_dependencies(SSJJNative ssjj_encrypt)
message(STATUS "Injector embeds encrypted native DLL: ${SSJJ_NATIVE_DLL_RC}")

# ---------------------------------------------------------------------------
# SSJJDrv.sys - kernel-mode injector (optional; requires WDK)
# The driver replaces the old CreateRemoteThread injection. It performs
# ZwOpenProcess/ZwAllocateVirtualMemory/ZwWriteVirtualMemory/
# ZwCreateThreadEx(LoadLibraryW) from SYSTEM context, bypassing
# GameGuard's user-mode API hooks. Must be signed (user provides cert).
# ---------------------------------------------------------------------------
set(WDK_ROOT "C:/Program Files (x86)/Windows Kits/10" CACHE PATH
    "WDK root directory (for building SSJJDrv.sys)")

# Locate the newest WDK km include directory (wdm.h)
set(WDK_KM_INCLUDE "" CACHE PATH "WDK km include dir (auto-detected)")
if(NOT WDK_KM_INCLUDE)
    file(GLOB WDK_VERSION_DIRS "${WDK_ROOT}/Include/*")
    list(SORT WDK_VERSION_DIRS ORDER DESCENDING)
    foreach(_dir IN LISTS WDK_VERSION_DIRS)
        get_filename_component(_ver "${_dir}" NAME)
        if(EXISTS "${_dir}/km/wdm.h")
            set(WDK_KM_INCLUDE "${_dir}/km")
            set(WDK_SHARED_INCLUDE "${_dir}/shared")
            set(WDK_UM_INCLUDE "${_dir}/um")
            set(WDK_UCRT_INCLUDE "${WDK_ROOT}/Include/${_ver}/ucrt")
            if(NOT EXISTS "${WDK_UCRT_INCLUDE}")
                # Some WDK builds ship no ucrt dir; fall back to the newest one
                file(GLOB _ucrt_candidates "${WDK_ROOT}/Include/*/ucrt")
                list(SORT _ucrt_candidates ORDER DESCENDING)
                if(_ucrt_candidates)
                    list(GET _ucrt_candidates 0 WDK_UCRT_INCLUDE)
                endif()
            endif()
            set(WDK_LIB_KM "${WDK_ROOT}/Lib/${_ver}/km/x64")
            set(WDK_LIB_UM "${WDK_ROOT}/Lib/${_ver}/um/x64")
            set(WDK_LIB_UCRT "${WDK_ROOT}/Lib/${_ver}/ucrt/x64")
            break()
        endif()
    endforeach()
endif()

if(WDK_KM_INCLUDE AND EXISTS "${WDK_KM_INCLUDE}/wdm.h")
    message(STATUS "WDK found: ${WDK_KM_INCLUDE}; building SSJJDrv.sys")
    add_library(SSJJDrv SHARED
        driver/SSJJDrv.c
        driver/SSJJProtect.c
        driver/SSJJDrv.h
        driver/SSJJProtect.h)
    set_target_properties(SSJJDrv PROPERTIES
        OUTPUT_NAME "SSJJDrv"
        SUFFIX ".sys"
        PREFIX "")

    target_include_directories(SSJJDrv PRIVATE
        ${CMAKE_CURRENT_SOURCE_DIR}/driver
        ${WDK_KM_INCLUDE}
        ${WDK_SHARED_INCLUDE}
        ${WDK_UM_INCLUDE}
        ${WDK_UCRT_INCLUDE})

    # WDK kernel-mode flags (MSVC); /utf-8: source has CJK comments
    target_compile_options(SSJJDrv PRIVATE
        /kernel /W3 /O2 /MT /Zp8 /GS /Gy /Zl /utf-8
        /D_WIN64 /D_AMD64_ /DAMD64 /DDRIVER
        /DNTDDI_VERSION=0x0A000007 /DWINVER=0x0A00 /D_WIN32_WINNT=0x0A00)

    target_link_options(SSJJDrv PRIVATE
        /DRIVER /SUBSYSTEM:NATIVE /ENTRY:DriverEntry
        /NODEFAULTLIB /SECTION:INIT,D /MACHINE:X64 /MANIFEST:NO)

    target_link_libraries(SSJJDrv PRIVATE
        ${WDK_LIB_KM}/ntoskrnl.lib
        ${WDK_LIB_KM}/hal.lib
        ${WDK_LIB_KM}/wdmsec.lib
        ${WDK_LIB_KM}/BufferOverflowFastFailK.lib)

    # Copy the signed driver next to Vape.exe (dist/)
    add_custom_command(TARGET SSJJDrv POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different
            $<TARGET_FILE:SSJJDrv>
            "${CMAKE_BINARY_DIR}/dist/SSJJDrv.sys"
        COMMENT "[DRV] SSJJDrv.sys -> dist/"
        VERBATIM)

    # The injector depends on the driver existing at runtime (beside exe)
    add_dependencies(SSJJInjector SSJJDrv)
else()
    message(WARNING
        "WDK not found (${WDK_ROOT}). SSJJDrv.sys will NOT be built.\n"
        "Install Windows Driver Kit or set -DWDK_ROOT=<WDK install dir>.\n"
        "The injector requires a signed SSJJDrv.sys beside Vape.exe.")
endif()
